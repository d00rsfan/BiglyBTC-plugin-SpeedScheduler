cmake_minimum_required(VERSION 3.25)
project(BiglyBTC-plugin-SpeedScheduler VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Toolchain.cmake")
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/Toolchain.cmake")
else()
  include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/Toolchain.cmake")
endif()
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core)
add_library(biglybtc_scheduler_policy STATIC scheduler.cpp original_xml.cpp ownership_file.cpp)
set_target_properties(biglybtc_scheduler_policy PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_include_directories(biglybtc_scheduler_policy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(biglybtc_scheduler_policy PRIVATE Qt6::Core)
target_compile_features(biglybtc_scheduler_policy PUBLIC cxx_std_23)
target_compile_options(biglybtc_scheduler_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(BIGLYBTC_BUILD_PLUGIN "Build independently loadable plugin module" ON)
if(BIGLYBTC_BUILD_PLUGIN)
  find_package(BiglyBTCPluginSDK 0.1.0 EXACT CONFIG REQUIRED)
  add_library(BiglyBTC-plugin-SpeedScheduler MODULE module.cpp)
  set_target_properties(BiglyBTC-plugin-SpeedScheduler PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON CXX_EXTENSIONS OFF)
  target_link_libraries(BiglyBTC-plugin-SpeedScheduler PRIVATE biglybtc_scheduler_policy BiglyBTC::PluginAPI)
  target_compile_options(BiglyBTC-plugin-SpeedScheduler PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_executable(scheduler-module-driver module_driver.cpp)
  target_link_libraries(scheduler-module-driver PRIVATE BiglyBTC::PluginHost)
  target_compile_options(scheduler-module-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(scheduler-tests tests.cpp)
target_link_libraries(scheduler-tests PRIVATE biglybtc_scheduler_policy Qt6::Core)
target_compile_options(scheduler-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME original-scheduler-policy COMMAND scheduler-tests)
set_tests_properties(original-scheduler-policy PROPERTIES TIMEOUT 60)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(BIGLYBTC_BUILD_PLUGIN)
  add_test(NAME original-scheduler-module-fixtures COMMAND Python3::Interpreter
    "${CMAKE_CURRENT_SOURCE_DIR}/module_fixture_test.py" "$<TARGET_FILE:scheduler-module-driver>"
    "$<TARGET_FILE:BiglyBTC-plugin-SpeedScheduler>")
  set_tests_properties(original-scheduler-module-fixtures PROPERTIES TIMEOUT 180)
endif()
add_test(NAME original-scheduler-xml-time-fixtures COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/fixture_test.py" "$<TARGET_FILE:scheduler-tests>")
set_tests_properties(original-scheduler-xml-time-fixtures PROPERTIES TIMEOUT 120)
add_test(NAME original-scheduler-tick-fixtures COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/tick_fixture_test.py" "$<TARGET_FILE:scheduler-tests>")
set_tests_properties(original-scheduler-tick-fixtures PROPERTIES TIMEOUT 180)
add_executable(ownership-tests ownership_tests.cpp)
target_link_libraries(ownership-tests PRIVATE biglybtc_scheduler_policy)
target_compile_options(ownership-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME original-pause-ownership COMMAND ownership-tests)
set_tests_properties(original-pause-ownership PROPERTIES TIMEOUT 60)
add_test(NAME original-pause-ownership-fixtures COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/ownership_fixture_test.py" "$<TARGET_FILE:ownership-tests>")
set_tests_properties(original-pause-ownership-fixtures PROPERTIES TIMEOUT 120)
